import { useInfiniteQuery } from "@tanstack/react-query";
import { useInView } from "react-intersection-observer";
import { useEffect } from "react";
import VideoCard from "@/components/library/VideoCard";
import api from "@/utils/api";

export default function MyShorts() {
  const { data, fetchNextPage, hasNextPage, isFetchingNextPage } =
    useInfiniteQuery({
      queryKey: ["myShorts"],
      queryFn: ({ pageParam }) =>
        api.get("/videos", { params: { cursor: pageParam } }).then((r) => r.data),
      initialPageParam: "",
      getNextPageParam: (last) => last.nextCursor,
    });

  const { ref, inView } = useInView();
  useEffect(() => {
    if (inView && hasNextPage && !isFetchingNextPage) fetchNextPage();
  }, [inView]);

  return (
    <main className="mx-auto max-w-md px-2 py-4">
      <h1 className="text-3xl font-bold mb-4 bg-brand-gradient bg-clip-text text-transparent">
        My Shorts
      </h1>

      <div className="grid gap-4">
        {data?.pages.flatMap((p) =>
          p.items.map((v: any) => <VideoCard key={v.id} video={v} />)
        )}
      </div>

      <div ref={ref} className="h-8" />
    </main>
  );
}

import api from "@/utils/api";

  const [job, setJob] = useState<any>(null);

      <PromptInput onGenerate={generate} loading={isPending} />

  );

interface Props {

  loading: boolean;

export default function PromptInput({ onGenerate, loading }: Props) {
  const [prompt, setPrompt] = useState("");

  return (
    <form
      onSubmit={(e) => {
        e.preventDefault();
        onGenerate(prompt);
      }}
      className="flex gap-2"
    >
      <input
        className="flex-1 rounded border px-3 py-1"
        placeholder="Describe your short..."
        value={prompt}
        onChange={(e) => setPrompt(e.target.value)}
      />
      <button
        type="submit"
        disabled={loading}
        className="rounded bg-brand-gradient px-4 py-1 text-white"
      >
        {loading ? "..." : "Generate"}
      </button>
    </form>
  );
}

import { useEffect } from "react";
import { useQuery } from "@tanstack/react-query";
import toast from "react-hot-toast";
import api from "@/utils/api";

export default function VeoPreview({ job }: { job: any }) {
  const { data: status } = useQuery({
    queryKey: ["veo", job.id],
    queryFn: () => api.get(`/veo/${job.id}`).then((r) => r.data),
    refetchInterval: (q) => (q.state.data?.done ? false : 2000),
  });

  if (!status) return <p>Waiting...</p>;
  if (status.failed) return <p className="text-red-500">Failed</p>;
  if (!status.done) return <p>Processing {status.progress}%</p>;

  return (
    <div className="mt-4">
      <video
        src={status.url}
        controls
        className="rounded aspect-[9/16] w-full"
      />
      <button
        onClick={async () => {
          await api.post(`/veo/${job.id}/save`);
          toast.success("Saved to My Shorts");
        }}
        className="mt-2 w-full rounded bg-brand-gradient py-2 text-white"
      >
        Save to My Shorts
      </button>
    </div>
  );
}

backend/src/services/videoService.ts
TypeScript

import { Storage } from "@google-cloud/storage";
import { getSignedUrl } from "@google-cloud/storage/build/esm/signer.js";
import { prisma } from "../utils/db.js";

const storage = new Storage();
const bucket = storage.bucket("shortshub-raw-uploads");

export const videoService = {
  async list(userId: string, cursor?: string) {
    const take = 20;
    const videos = await prisma.video.findMany({
      where: { userId },
      orderBy: { createdAt: "desc" },
      take: take + 1,
      cursor: cursor ? { id: cursor } : undefined,
    });
    const nextCursor = videos.length > take ? videos.pop()?.id : null;
    return { items: videos, nextCursor };
  },

  async getUploadUrl(userId: string, filename: string, contentType: string) {
    const file = bucket.file(`${userId}/${crypto.randomUUID()}_${filename}`);
    const [url] = await file.getSignedUrl({
      version: "v4",
      action: "resumable",
      method: "POST",
      contentType,
      expires: Date.now() + 15 * 60 * 1000,
    });
    return url;
  },
};

import { Router } from "express";
import { PubSub } from "@google-cloud/pubsub";
import { auth } from "../middleware/auth.js";
import { prisma } from "../utils/db.js";

const r = Router();
r.use(auth);

const pubsub = new PubSub();

r.post("/", async (req, res) => {
  const { prompt, count = 1 } = req.body;
  const jobId = crypto.randomUUID();
  await pubsub.topic("veo-generator").publishMessage({
    json: { prompt, userId: req.user!.id, jobId, count },
  });
  res.json({ id: jobId });
});

r.get("/:id", async (req, res) => {
  const job = await prisma.video.findUnique({
    where: { id: req.params.id },
  });
  if (!job) return res.status(404).send("not found");
  res.json({ done: job.status === "ready", progress: 100, url: job.variants?.temp });
});

r.post("/:id/save", async (req, res) => {
  await prisma.video.update({
    where: { id: req.params.id },
    data: { status: "ready" },
  });
  res.sendStatus(204);
});

export default r;

import { Storage } from "@google-cloud/storage";
import ffmpeg from "fluent-ffmpeg";
import ffmpegPath from "ffmpeg-static";
import { tmpdir } from "os";
import { join } from "path";
import { createWriteStream } from "fs";

ffmpeg.setFfmpegPath(ffmpegPath);

const storage = new Storage();
const srcBucket = storage.bucket("shortshub-raw-uploads");
const dstBucket = storage.bucket("shortshub-processed-variants");

export async function reformat(event, ctx) {
  const file = srcBucket.file(event.name);
  const tmpFile = join(tmpdir(), "in.mp4");
  await file.createReadStream().pipe(createWriteStream(tmpFile));

  const outFile = join(tmpdir(), "out.mp4");
  await new Promise((res, rej) =>
    ffmpeg(tmpFile)
      .outputOptions("-vf", "scale=1080:1920", "-c:v", "libx264", "-c:a", "aac")
      .save(outFile)
      .on("end", res)
      .on("error", rej)
  );

  const dstName = `processed_${ctx.eventId}.mp4`;
  await dstBucket.upload(outFile, { destination: dstName, public: true });
  const url = `https://storage.googleapis.com/${dstBucket.name}/${dstName}`;

  // TODO: update DB row with url
}

import { GoogleGenerativeAI } from "@google/generative-ai";
import { Storage } from "@google-cloud/storage";
import { PubSub } from "@google-cloud/pubsub";

const genAI = new GoogleGenerativeAI(process.env.GEMINI_KEY);
const model = genAI.getGenerativeModel({ model: "veo" });
const storage = new Storage();
const tmpBucket = storage.bucket("shortshub-veo-outputs-temp");
const pubsub = new PubSub();

export async function handler(message, ctx) {
  const { prompt, userId, jobId } = JSON.parse(Buffer.from(message.data, "base64").toString());

  const result = await model.generateContent({
    prompt,
    outputConfig: { format: "mp4" },
  });

  const url = result.video.url;
  const file = tmpBucket.file(`${userId}/${jobId}.mp4`);
  await file.save(Buffer.from(await (await fetch(url)).arrayBuffer()), { public: true });
  const publicUrl = file.publicUrl();

  // update DB
  await pubsub.topic("veo-status").publish(Buffer.from(JSON.stringify({ jobId, url: publicUrl })));
}

import { PrismaClient } from "@prisma/client";
import { google } from "googleapis";
import { Storage } from "@google-cloud/storage";

const prisma = new PrismaClient();
const storage = new Storage();

export async function handler() {
  const posts = await prisma.scheduledPost.findMany({
    where: { status: "pending", fireAt: { lte: new Date() } },
    include: { video: true },
  });

  for (const p of posts) {
    try {
      const yt = p.platforms.youtube;
      if (yt) {
        const oauth2 = new google.auth.OAuth2();
        oauth2.setCredentials({ access_token: yt.token });
        const youtube = google.youtube({ version: "v3", auth: oauth2 });
        await youtube.videos.insert({
          part: ["snippet", "status"],
          requestBody: {
            snippet: { title: yt.title, description: yt.desc, tags: yt.tags },
            status: { privacyStatus: yt.privacy },
          },
          media: { body: storage.bucket("shortshub-processed-variants").file(p.video.variants.youtube).createReadStream() },
        });
      }

      await prisma.scheduledPost.update({ where: { id: p.id }, data: { status: "posted" } });
    } catch (e) {
      console.error(e);
      await prisma.scheduledPost.update({ where: { id: p.id }, data: { status: "error" } });
    }
  }
}

# ShortsHub.app – MVP

## Quick start (local)

1. `git clone …`
2. `cd backend && cp .env.example .env` – fill in `DATABASE_URL`, `GEMINI_KEY`, etc.
3. `pnpm i && pnpm prisma migrate dev`
4. `pnpm dev`
5. `cd ../frontend && pnpm i && pnpm dev`

## Deploy

- Backend: `gcloud run deploy shortshub-api --source .`
- Frontend: `vercel --prod`
- Cloud Functions: `gcloud functions deploy …` (see `cloud-functions/*/README.md`)
Everything else (auth, scheduler UI, token refresh, etc.) follows the same patterns and can be added in <1 day without breaking the current contracts.
